cmake_minimum_required(VERSION 3.24)
# Standalone CLion metadata only. Actual builds belong to b; no fetch or link.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(hotcwap_ide LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Local Vexspoke src; never downloaded")
set(GRAPHVEX_SOURCE_DIR "" CACHE PATH "Local Graphvex src; never downloaded")
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "capability/*.[ch]" "hot/*.[ch]"
    "kernel/*.[ch]" "permission/*.[ch]" "spoke/*.[ch]" "window/*.[ch]")
if(APPLE)
    enable_language(OBJC)
    file(GLOB_RECURSE objc_sources CONFIGURE_DEPENDS "window/*.m" "permission/*.m")
    list(APPEND sources ${objc_sources})
    list(FILTER sources EXCLUDE REGEX "/window(_linux|_wayland|_win32)?\\.c$")
elseif(WIN32)
    list(FILTER sources EXCLUDE REGEX "/window(_linux|_wayland)?\\.c$")
else()
    list(FILTER sources EXCLUDE REGEX "/window_win32\\.c$")
endif()
add_library(hotcwap_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(hotcwap_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
foreach(path VEXSPOKE_SOURCE_DIR GRAPHVEX_SOURCE_DIR)
    if(${path})
        target_include_directories(hotcwap_ide PRIVATE "${${path}}")
    endif()
endforeach()
if(WIN32)
    target_compile_definitions(hotcwap_ide PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(hotcwap_ide PRIVATE -std=gnu23 -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(hotcwap_ide PRIVATE -mcpu=apple-m1 "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc>")
    endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    execute_process(COMMAND "${CMAKE_C_COMPILER}" -print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(EXISTS "${resource}/include/stddef.h")
        target_compile_options(hotcwap_ide PRIVATE -isystem "${resource}/include")
    endif()
endif()
message(STATUS "IDE metadata only: excluded object target, host backend only. Use b for builds; no linking or runner.")
